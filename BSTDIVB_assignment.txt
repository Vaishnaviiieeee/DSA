class Node:
    def __init__(self, score):
        self.score, self.left, self.right = score, None, None

class BST:
    def __init__(self):
        self.root = None

    def insert(self, root, val):
        if not root:
            return Node(val)
        if val < root.score:
            root.left = self.insert(root.left, val)
        elif val > root.score:
            root.right = self.insert(root.right, val)
        return root

    def inorder(self, root):
        if not root:
            return []
        return self.inorder(root.left) + [root.score] + self.inorder(root.right)

    def min_val(self, root):
        while root and root.left:
            root = root.left
        return root.score if root else None

    def max_val(self, root):
        while root and root.right:
            root = root.right
        return root.score if root else None

    def count_above(self, root, x=50):
        if not root:
            return 0
        c = 1 if root.score > x else 0
        return c + self.count_above(root.left, x) + self.count_above(root.right, x)

    def search(self, root, target):
        if not root or root.score == target:
            return root is not None
        return self.search(root.left, target) if target < root.score else self.search(root.right, target)

bst = BST()
scores = [45, 78, 32, 89, 56, 92, 67, 34, 85, 73, 61, 98]
for s in scores:
    bst.root = bst.insert(bst.root, s)

asc = bst.inorder(bst.root)
print("Ascending:", asc)
print("Highest:", bst.max_val(bst.root), "| Lowest:", bst.min_val(bst.root))
print("Above 50:", bst.count_above(bst.root, 50))
q = int(input("Search score: "))
print("Found" if bst.search(bst.root, q) else "Not found")
print("Average:", round(sum(asc) / len(asc), 2))
